#!/usr/bin/env python3
"""Load distilled JSONL into ChromaDB for semantic search."""

import json
import chromadb
from pathlib import Path

VECTORS_DIR = Path("/root/gcp-cloud-distill/output/vectors")
CHROMA_DIR = Path("/root/gcp-cloud-distill/chromadb_data")

def load_jsonl(path):
    records = []
    with open(path, "r", encoding="utf-8") as f:
        for line in f:
            if line.strip():
                records.append(json.loads(line))
    return records

def main():
    client = chromadb.PersistentClient(path=str(CHROMA_DIR))

    # Load each channel as a separate collection
    channels = {
        "all": "all_knowledge.jsonl",
        "persona": "persona_knowledge.jsonl",
        "tech_stack": "tech_stack_knowledge.jsonl",
        "timeline": "timeline_knowledge.jsonl",
    }

    for name, filename in channels.items():
        filepath = VECTORS_DIR / filename
        if not filepath.exists():
            print(f"Skip {name}: {filepath} not found")
            continue

        records = load_jsonl(filepath)
        if not records:
            print(f"Skip {name}: empty")
            continue

        # Create or get collection (ChromaDB auto-embeds with built-in model)
        collection = client.get_or_create_collection(
            name=f"cloud_distill_{name}",
            metadata={"hnsw:space": "cosine"}
        )

        # Batch insert
        ids = [r["id"] for r in records]
        documents = [r["content"] for r in records]
        metadatas = [r.get("metadata", {}) for r in records]

        # ChromaDB has a batch limit of 5461
        batch_size = 5000
        for i in range(0, len(ids), batch_size):
            batch_ids = ids[i:i+batch_size]
            batch_docs = documents[i:i+batch_size]
            batch_meta = metadatas[i:i+batch_size]
            # Sanitize metadata (ChromaDB only accepts str/int/float/bool)
            clean_meta = []
            for m in batch_meta:
                clean = {}
                for k, v in m.items():
                    if isinstance(v, (str, int, float, bool)):
                        clean[k] = v
                    elif isinstance(v, list):
                        clean[k] = ", ".join(str(x) for x in v)
                    else:
                        clean[k] = str(v)
                clean_meta.append(clean)
            collection.upsert(ids=batch_ids, documents=batch_docs, metadatas=clean_meta)

        print(f"  {name}: loaded {len(ids)} records into ChromaDB")

    print(f"\nChromaDB data stored at: {CHROMA_DIR}")
    print("Query example:")
    print('  collection = client.get_collection("cloud_distill_all")')
    print('  results = collection.query(query_texts=["firebase deployment"], n_results=5)')

if __name__ == "__main__":
    main()
PYEOF

================================================================================
TROUBLESHOOTING
================================================================================

Q: rclone says "token expired"
A: On your LOCAL Mac, run: rclone authorize onedrive (or drive)
   Copy the new token JSON, paste into VPS rclone.conf

Q: Gemini API returns 429 (rate limit)
A: Increase delay_seconds in config.yaml (e.g. 2.0)

Q: Some DOCX/PDF files fail to parse
A: Normal. The parser logs failures. Check parsed/*.json for error details.

Q: Pipeline stops mid-way
A: It saves checkpoints every 20 docs. Just re-run the same step.

Q: How to distill with Grok instead of Gemini?
A: Edit config.yaml: distill.provider: "grok"
   Set GROK_API_KEY in .env. Note: Grok costs $3/M input + $15/M output.
